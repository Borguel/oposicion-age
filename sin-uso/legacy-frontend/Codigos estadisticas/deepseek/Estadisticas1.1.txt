<script>
document.addEventListener("DOMContentLoaded", async function () {
  const email = window.usuarioEmail || "test@ejemplo.com";
  const refreshBtn = document.getElementById("estadisticas-refresh");
  const modal = document.getElementById("modal-temas");
  const modalCerrar = document.querySelector(".modal-cerrar");
  const btnVerNuevos = document.getElementById("btn-ver-nuevos");
  
  // Función para cargar datos
  async function cargarDatos() {
    try {
      // Mostrar estado de carga
      document.querySelectorAll('.valor').forEach(el => {
        el.textContent = '...';
      });
      
      refreshBtn.disabled = true;
      refreshBtn.innerHTML = `
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg" style="animation: spin 1s linear infinite;">
          <path d="M12 4V8L16 4L12 0V4ZM12 20V16L8 20L12 24V20Z" fill="white"/>
        </svg>
        Cargando...
      `;
      
      const [resumenRes, temasRes] = await Promise.all([
        fetch(`https://oposicion-age.onrender.com/resumen-progreso?usuario_id=${email}`),
        fetch("https://oposicion-age.onrender.com/temas-disponibles")
      ]);
      
      const resumenData = await resumenRes.json();
      console.log("Datos recibidos:", resumenData);
      const temasData = await temasRes.json();

      const resumen = resumenData.resumen ?? {};
      const totalTests = resumen.tests_realizados ?? 0;
      const totalAciertos = resumen.total_aciertos ?? 0;
      const totalFallos = resumen.total_fallos ?? 0;
      const historial = resumen.historial_tests ?? [];
      const temasTest = resumen.temas_test ?? [];
      const esquemas = resumen.esquemas_realizados ?? 0;
      const aprobados = resumen.tests_aprobados ?? 0;
      const suspendidos = resumen.tests_suspendidos ?? 0;
      const puntuacionMedia = resumen.puntuacion_media_test ?? 0;

      // Tiempo total
      const totalSeg = resumen.tiempo_total ?? 0;
      const horas = String(Math.floor(totalSeg / 3600)).padStart(2, '0');
      const minutos = String(Math.floor((totalSeg % 3600) / 60)).padStart(2, '0');
      
      // Porcentajes
      const porcentajeAprobados = totalTests > 0 ? Math.round((aprobados / totalTests) * 100) : 0;
      const porcentajeSuspendidos = totalTests > 0 ? Math.round((suspendidos / totalTests) * 100) : 0;
      const porcentajeAciertos = (totalAciertos + totalFallos) > 0 
        ? Math.round((totalAciertos / (totalAciertos + totalFallos)) * 100) 
        : 0;
      const porcentajeFallos = (totalAciertos + totalFallos) > 0 
        ? Math.round((totalFallos / (totalAciertos + totalFallos)) * 100) 
        : 0;

      // Temas más estudiados
      const contador = {};
      historial.forEach(test => {
        (test.temas || []).forEach(t => {
          contador[t] = (contador[t] || 0) + 1;
        });
      });
      
      const topTemasEntries = Object.entries(contador)
        .sort((a, b) => b[1] - a[1])
        .slice(0, 3);
      
      // Temas no estudiados
      const todosLosTemas = (temasData.temas || []).map(t => t.id);
      const noEstudiados = todosLosTemas.filter(t => !temasTest.includes(t));
      const noEstudiadosStr = noEstudiados.length ? noEstudiados.length : "0";
      
      const noEstudiadosNombres = noEstudiados
        .map(id => {
          const tema = (temasData.temas || []).find(t => t.id === id);
          return tema ? tema.titulo : id;
        });

      // Mostrar en HTML
      document.getElementById("tests").textContent = totalTests;
      document.getElementById("aprobados").textContent = aprobados;
      document.getElementById("suspendidos").textContent = suspendidos;
      document.getElementById("media").textContent = puntuacionMedia.toFixed(1);
      document.getElementById("aciertos").textContent = totalAciertos;
      document.getElementById("fallos").textContent = totalFallos;
      document.getElementById("esquemas").textContent = esquemas;
      document.getElementById("tiempo").textContent = `${horas}h ${minutos}m`;
      document.getElementById("temas-nuevos").textContent = noEstudiadosStr;
      
      // Actualizar lista de temas más estudiados
      actualizarTemas(topTemasEntries, temasData.temas || []);
      
      // Porcentajes
      document.getElementById("aprobados-porcentaje").textContent = porcentajeAprobados;
      document.getElementById("suspendidos-porcentaje").textContent = porcentajeSuspendidos;
      document.getElementById("aciertos-porcentaje").textContent = porcentajeAciertos;
      document.getElementById("fallos-porcentaje").textContent = porcentajeFallos;
      
      // Guardar datos para el modal
      document.getElementById("temas-nuevos-container").dataset.temas = JSON.stringify(noEstudiados);
      document.getElementById("temas-nuevos-container").dataset.todosTemas = JSON.stringify(temasData.temas || []);
      
    } catch (err) {
      console.error("Error cargando estadísticas:", err);
    } finally {
      refreshBtn.disabled = false;
      refreshBtn.innerHTML = `
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg">
          <path d="M17.65 6.35C16.2 4.9 14.21 4 12 4C7.58 4 4 7.58 4 12C4 16.42 7.58 20 12 20C15.73 20 18.84 17.45 19.73 14H17.65C16.83 16.33 14.61 18 12 18C8.69 18 6 15.31 6 12C6 8.69 8.69 6 12 6C13.66 6 15.14 6.69 16.22 7.78L13 11H20V4L17.65 6.35Z" fill="white"/>
        </svg>
        Actualizar
      `;
    }
  }
  
  // Función para actualizar la lista de temas
  function actualizarTemas(temasEntries, todosTemas) {
    const listaTemas = document.getElementById("lista-temas-top");
    listaTemas.innerHTML = '';
    
    temasEntries.forEach(([id, count]) => {
      const tema = todosTemas.find(t => t.id === id);
      const nombre = tema ? tema.titulo : id;
      
      const li = document.createElement('li');
      li.textContent = `${nombre} (${count} tests)`;
      listaTemas.appendChild(li);
    });
  }
  
  // Función para mostrar el modal con temas no estudiados
  function mostrarModalTemasNoEstudiados(temasIds, todosTemas) {
    const listaTemas = document.getElementById("lista-temas-nuevos");
    listaTemas.innerHTML = '';
    
    temasIds.forEach(id => {
      const tema = todosTemas.find(t => t.id === id);
      const nombre = tema ? tema.titulo : id;
      
      const li = document.createElement('li');
      li.textContent = nombre;
      listaTemas.appendChild(li);
    });
    
    modal.style.display = 'flex';
  }
  
  // Configurar eventos para mostrar/ocultar temas más estudiados
  document.getElementById('temas-top-container').addEventListener('mouseenter', function() {
    this.classList.add('activo');
  });
  
  document.getElementById('temas-top-container').addEventListener('mouseleave', function() {
    this.classList.remove('activo');
  });
  
  document.querySelector('#temas-top-container .btn-ver-temas').addEventListener('click', function(e) {
    e.stopPropagation();
    this.closest('.tarjeta-temas').classList.toggle('activo');
  });
  
  // Configurar evento para el modal de temas no estudiados
  btnVerNuevos.addEventListener('click', function() {
    const temasIds = JSON.parse(this.closest('.tarjeta-estadistica').dataset.temas || '[]');
    const todosTemas = JSON.parse(this.closest('.tarjeta-estadistica').dataset.todosTemas || '[]');
    mostrarModalTemasNoEstudiados(temasIds, todosTemas);
  });
  
  // Cerrar modal al hacer clic en la X
  modalCerrar.addEventListener('click', () => {
    modal.style.display = 'none';
  });
  
  // Cerrar modal al hacer clic fuera del contenido
  modal.addEventListener('click', (e) => {
    if (e.target === modal) {
      modal.style.display = 'none';
    }
  });
  
  // Configurar botón de refrescar
  refreshBtn.addEventListener('click', cargarDatos);
  
  // Cargar datos iniciales
  cargarDatos();
  
  // Animación para el icono de carga
  const style = document.createElement('style');
  style.textContent = `
    @keyframes spin {
      from { transform: rotate(0deg); }
      to { transform: rotate(360deg); }
    }
  `;
  document.head.appendChild(style);
});
</script>
