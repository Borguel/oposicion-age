    <script>
        // ✅ Usa tu URL de Render
        const BACKEND_URL = "https://oposicion-age.onrender.com";
        
        // Elementos del DOM
        const uploadArea = document.getElementById('upload-area');
        const pdfFileInput = document.getElementById('pdf-file');
        const selectPdfBtn = document.getElementById('select-pdf-btn');
        const generateBtnContainer = document.getElementById('generate-btn-container');
        const generateBtn = document.getElementById('generate-btn');
        const resumenContainer = document.getElementById('resumen-container');
        const resumenContent = document.getElementById('resumen-content');
        const copyBtn = document.getElementById('copy-btn');
        const chatBox = document.getElementById('chat-box');
        const userInput = document.getElementById('user-input');
        const sendBtn = document.getElementById('send-btn');
        
        let currentPdfFile = null;
        
        // Evento para abrir el selector de archivos
        selectPdfBtn.addEventListener('click', () => {
            pdfFileInput.click();
        });
        
        // Evento para arrastrar y soltar archivos
        uploadArea.addEventListener('dragover', (e) => {
            e.preventDefault();
        });
        
        uploadArea.addEventListener('drop', (e) => {
            e.preventDefault();
            
            if (e.dataTransfer.files.length) {
                pdfFileInput.files = e.dataTransfer.files;
                handleFileSelect();
            }
        });
        
        // Evento para cuando se selecciona un archivo
        pdfFileInput.addEventListener('change', handleFileSelect);
        
        // Manejar la selección de archivos
        function handleFileSelect() {
            const file = pdfFileInput.files[0];
            
            if (!file) {
                alert('Por favor selecciona un archivo PDF');
                return;
            }
            
            currentPdfFile = file;
            
            // Mostrar el nombre del archivo
            const fileName = file.name.length > 30 ? file.name.substring(0, 27) + '...' : file.name;
            uploadArea.innerHTML = `
                <div class="upload-icon">
                    <i class="fas fa-file-pdf"></i>
                </div>
                <div class="upload-text">
                    <h3>Archivo seleccionado</h3>
                    <p>${fileName}</p>
                </div>
                <button class="btn" id="change-file-btn">
                    <i class="fas fa-sync"></i> Cambiar archivo
                </button>
            `;
            
            // Mostrar botón de generar resumen
            generateBtnContainer.style.display = 'block';
            
            // Añadir evento al botón de cambiar archivo
            document.getElementById('change-file-btn').addEventListener('click', () => {
                pdfFileInput.click();
            });
        }
        
        // Generar resumen al hacer clic en el botón
        generateBtn.addEventListener('click', generateSummary);
        
        // Función para generar resumen
        async function generateSummary() {
            if (!currentPdfFile) return;
            
            const formData = new FormData();
            formData.append('pdf', currentPdfFile);
            
            try {
                // Mostrar mensaje de espera
                generateBtn.innerHTML = '<div class="spinner"></div> Generando...';
                generateBtn.disabled = true;
                
                const response = await fetch(`${BACKEND_URL}/resumir-pdf`, {
                    method: 'POST',
                    body: formData
                });
                
                const data = await response.json();
                
                generateBtn.innerHTML = '<i class="fas fa-file-contract"></i> Generar Resumen';
                generateBtn.disabled = false;
                
                if (data.resumen) {
                    // Formatear el resumen con saltos de línea
                    const formattedResumen = data.resumen.replace(/\n/g, '<br>');
                    resumenContent.innerHTML = formattedResumen;
                    resumenContainer.style.display = 'flex';
                    
                    // Añadir mensaje al chat
                    addMessageToChat("He generado un resumen del documento PDF.", 'user');
                    
                    // Simular respuesta del asistente
                    setTimeout(() => {
                        addMessageToChat("He creado un resumen del PDF que subiste. ¿Te gustaría que te explique algún punto en particular o que responda preguntas sobre este contenido?", 'bot');
                    }, 1000);
                } else {
                    alert("Error: " + (data.error || 'Error al generar el resumen'));
                }
            } catch (error) {
                generateBtn.innerHTML = '<i class="fas fa-file-contract"></i> Generar Resumen';
                generateBtn.disabled = false;
                alert("Error de conexión: " + error.message);
            }
        }
        
        // Copiar resumen al portapapeles
        copyBtn.addEventListener('click', () => {
            const textToCopy = resumenContent.innerText;
            navigator.clipboard.writeText(textToCopy).then(() => {
                const originalText = copyBtn.innerHTML;
                copyBtn.innerHTML = '<i class="fas fa-check"></i> Copiado!';
                setTimeout(() => {
                    copyBtn.innerHTML = originalText;
                }, 2000);
            });
        });
        
        // Función para enviar mensajes
        function enviarMensaje() {
            const message = userInput.value.trim();
            
            if (!message) return;
            
            // Añadir mensaje de usuario al chat
            addMessageToChat(message, 'user');
            
            // Limpiar el input
            userInput.value = '';
            
            // Mostrar estado de "escribiendo"
            showTypingIndicator();
            
            // Simular respuesta después de un retraso
            setTimeout(() => {
                // Ocultar indicador de escritura
                hideTypingIndicator();
                
                // Generar respuesta basada en el mensaje del usuario
                let response = "Gracias por tu pregunta. Estoy procesando tu consulta y te responderé en breve.";
                
                if (message.toLowerCase().includes('resumen')) {
                    response = "Para generar un resumen, primero sube un documento PDF usando el panel de la izquierda y luego haz clic en 'Generar Resumen'.";
                } else if (message.toLowerCase().includes('ayuda')) {
                    response = "Puedo ayudarte a:<br>- Resumir documentos PDF<br>- Responder preguntas sobre el temario<br>- Explicar conceptos complejos<br>- Prepararte para los exámenes";
                } else {
                    response = "He procesado tu pregunta sobre \"" + message + "\". Como asistente especializado en oposiciones para Administrativo del Estado, puedo darte información precisa sobre este tema. ¿Quieres que profundice en algún aspecto específico?";
                }
                
                addMessageToChat(response, 'bot');
            }, 1500);
        }
        
        // Función para añadir mensajes al chat
        function addMessageToChat(message, sender) {
            const messageElement = document.createElement('div');
            messageElement.classList.add('message');
            messageElement.classList.add(sender === 'user' ? 'user-message' : 'bot-message');
            
            const now = new Date();
            const timeString = now.getHours() + ':' + (now.getMinutes() < 10 ? '0' : '') + now.getMinutes();
            
            messageElement.innerHTML = `
                <div class="message-header">
                    <i class="fas ${sender === 'user' ? 'fa-user' : 'fa-robot'}"></i>
                    <span>${sender === 'user' ? 'Tú' : 'Asistente IA'}</span>
                </div>
                <div class="message-content">
                    ${message}
                </div>
                <div class="message-time">${timeString}</div>
            `;
            
            chatBox.appendChild(messageElement);
            chatBox.scrollTop = chatBox.scrollHeight;
        }
        
        // Función para mostrar indicador de "escribiendo"
        function showTypingIndicator() {
            const typingElement = document.createElement('div');
            typingElement.classList.add('message', 'bot-message');
            typingElement.id = 'typing-indicator';
            typingElement.innerHTML = `
                <div class="message-header">
                    <i class="fas fa-robot"></i>
                    <span>Asistente IA</span>
                </div>
                <div class="message-content">
                    <div class="spinner"></div> <span>Escribiendo...</span>
                </div>
            `;
            
            chatBox.appendChild(typingElement);
            chatBox.scrollTop = chatBox.scrollHeight;
        }
        
        // Función para ocultar indicador de "escribiendo"
        function hideTypingIndicator() {
            const typingElement = document.getElementById('typing-indicator');
            if (typingElement) {
                typingElement.remove();
            }
        }
        
        // Event listeners
        sendBtn.addEventListener('click', enviarMensaje);
        userInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                enviarMensaje();
            }
        });
    </script>
