        let currentAction = '';
        let selectedFile = null;
        let processedData = null;

        function openModal(action) {
            currentAction = action;
            const modal = document.getElementById('pdfModal');
            const title = document.getElementById('modalTitle');
            
            // Configurar título según la acción
            const titles = {
                'resumir': 'Resumir Documento PDF',
                'esquema': 'Generar Esquema desde PDF', 
                'test': 'Crear Test desde PDF',
                'tarjetas': 'Generar Tarjetas de Memoria'
            };
            
            title.textContent = titles[action] || 'Subir Documento PDF';
            modal.style.display = 'flex';
            
            // Resetear estado
            resetModal();
        }

        function closeModal() {
            document.getElementById('pdfModal').style.display = 'none';
            resetModal();
        }

        function resetModal() {
            selectedFile = null;
            document.getElementById('pdfFile').value = '';
            document.getElementById('fileName').textContent = '';
            document.getElementById('processBtn').disabled = true;
            document.getElementById('loading').style.display = 'none';
            document.getElementById('result').style.display = 'none';
            document.getElementById('resultContent').innerHTML = '';
        }

        function handleFileSelect(input) {
            if (input.files.length > 0) {
                selectedFile = input.files[0];
                document.getElementById('fileName').textContent = selectedFile.name;
                document.getElementById('processBtn').disabled = false;
            }
        }

        function closeResult() {
            document.getElementById('result').style.display = 'none';
        }

        async function processDocument() {
            if (!selectedFile) return;

            const loading = document.getElementById('loading');
            const processBtn = document.getElementById('processBtn');
            const resultDiv = document.getElementById('result');
            const resultContent = document.getElementById('resultContent');
            const resultTitle = document.getElementById('resultTitle');

            // Mostrar loading
            loading.style.display = 'block';
            processBtn.disabled = true;
            resultDiv.style.display = 'none';

            try {
                // Simular procesamiento (en un caso real, aquí iría la llamada a tu API)
                await new Promise(resolve => setTimeout(resolve, 2000));
                
                // Datos de ejemplo según el tipo de acción
                processedData = generateSampleData(currentAction);
                
                // Mostrar resultado
                resultTitle.textContent = getResultTitle(currentAction);
                resultContent.innerHTML = formatResult(processedData, currentAction);
                resultDiv.style.display = 'block';

            } catch (error) {
                console.error('Error:', error);
                resultContent.innerHTML = `<p style="color: red; text-align: center;">Error al procesar el documento: ${error.message}</p>`;
                resultDiv.style.display = 'block';
            } finally {
                loading.style.display = 'none';
            }
        }

        function getResultTitle(action) {
            const titles = {
                'resumir': 'Resumen del Documento',
                'esquema': 'Esquema Generado', 
                'test': 'Test Creado',
                'tarjetas': 'Tarjetas de Memoria'
            };
            return titles[action] || 'Resultado';
        }

        function formatResult(data, action) {
            switch(action) {
                case 'resumir':
                    return `<div class="resumen-content">${data.texto.replace(/\n/g, '</p><p>')}</div>`;
                
                case 'esquema':
                    return `<div class="esquema-content">${data.texto.replace(/\n/g, '</p><p>')}</div>`;
                
                case 'test':
                    if (data.preguntas && data.preguntas.length > 0) {
                        let html = '<div class="test-container">';
                        data.preguntas.forEach((pregunta, index) => {
                            html += `
                                <div class="pregunta-test">
                                    <div class="pregunta-texto">${index + 1}. ${pregunta.pregunta}</div>
                                    <div class="opciones-test">
                            `;
                            
                            Object.entries(pregunta.opciones).forEach(([key, value]) => {
                                html += `
                                    <div class="opcion-test">
                                        <input type="radio" id="p${index}_${key}" name="pregunta_${index}" value="${key}">
                                        <label for="p${index}_${key}">${key}) ${value}</label>
                                    </div>
                                `;
                            });
                            
                            html += `</div></div>`;
                        });
                        
                        html += `
                            <div class="test-actions">
                                <button class="btn-test secondary" onclick="closeResult()">Volver</button>
                                <button class="btn-test primary" onclick="iniciarTest()">Iniciar Test</button>
                            </div>
                        </div>`;
                        
                        return html;
                    }
                    return '<p>No se pudieron generar preguntas</p>';
                
                case 'tarjetas':
                    if (data.tarjetas && data.tarjetas.length > 0) {
                        let html = '<div class="flashcards-container">';
                        data.tarjetas.forEach((tarjeta, index) => {
                            html += `
                                <div class="flashcard" onclick="this.classList.toggle('flipped')">
                                    <div class="flashcard-number">${index + 1}</div>
                                    <div class="flashcard-front">
                                        ${tarjeta.pregunta}
                                    </div>
                                    <div class="flashcard-back">
                                        ${tarjeta.respuesta}
                                    </div>
                                </div>
                            `;
                        });
                        html += '</div>';
                        return html;
                    }
                    return '<p>No se pudieron generar tarjetas</p>';
                
                default:
                    return '<p>Resultado no formateado</p>';
            }
        }

        function downloadResult() {
            const { jsPDF } = window.jspdf;
            const doc = new jsPDF();
            const resultContent = document.getElementById('resultContent');
            const resultTitle = document.getElementById('resultTitle').textContent;
            
            // Obtener texto del resultado (simplificado)
            let text = resultContent.innerText || resultContent.textContent;
            
            // Configurar el PDF
            doc.setFontSize(16);
            doc.text(resultTitle, 20, 20);
            doc.setFontSize(12);
            
            // Dividir el texto en líneas
            const lines = doc.splitTextToSize(text, 170);
            doc.text(lines, 20, 40);
            
            // Guardar el PDF
            doc.save(`${resultTitle.toLowerCase().replace(/\s+/g, '_')}.pdf`);
        }
