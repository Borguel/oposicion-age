      Cargando test anterior...
    </div>

  <script src="https://cdn.jsdelivr.net/npm/sweetalert2@11"></script>
  <script>
  const usuarioEmail = window.usuarioEmail || "usuario_prueba";

  let preguntas = [];
  let respuestasUsuario = [];
  let indicePreguntaActual = 0;
  let tiempoInicio;
  let intervaloTemporizador;
  let temasTest = new Set();

  function iniciarTemporizador() {
    tiempoInicio = Date.now();
    intervaloTemporizador = setInterval(() => {
      const transcurrido = Math.floor((Date.now() - tiempoInicio) / 1000);
      const m = String(Math.floor(transcurrido / 60)).padStart(2, '0');
      const s = String(transcurrido % 60).padStart(2, '0');
      document.getElementById("temporizador").innerHTML = `
        <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <circle cx="12" cy="12" r="10"></circle>
          <polyline points="12 6 12 12 16 14"></polyline>
        </svg>
        <span>Tiempo: ${m}:${s}</span>
      `;
    }, 1000);
  }

  function mostrarPregunta(i) {
    // Actualizar barra de progreso
    const porcentajeProgreso = ((i + 1) / preguntas.length) * 100;
    document.getElementById("barra-progreso").style.width = porcentajeProgreso + "%";
    
    const p = preguntas[i];
    let html = `
      <div class="contador-pregunta">${i + 1}/${preguntas.length}</div>
      <div class="pregunta-texto">${p.pregunta}</div>
      <form id="form-pregunta">
        <div class="opciones-container">`;
    
    for (const letra in p.opciones) {
      const opcion = p.opciones[letra];
      const checked = respuestasUsuario[i] === letra ? "checked" : "";
      html += `
        <div class="opcion">
          <input type="radio" id="opcion-${letra}" name="respuesta" value="${letra}" ${checked}>
          <label for="opcion-${letra}">${opcion}</label>
        </div>`;
    }
    
    html += `</div>`;
    
    // Botones de navegación
    html += `
      <div class="botones-navegacion">
        <div>
          <button type="button" id="btn-desmarcar" class="btn btn-desmarcar">❌ Desmarcar respuesta</button>
        </div>
        <div style="display: flex; gap: 10px;">
          ${i > 0 ? `<button type="button" id="btn-anterior" class="btn btn-anterior">⬅️ Anterior</button>` : ''}
          <button type="submit" class="btn ${i === preguntas.length - 1 ? 'btn-finalizar' : 'btn-siguiente'}">
            ${i === preguntas.length - 1 ? 'Finalizar Test' : 'Siguiente ➡️'}
          </button>
        </div>
      </div>
    </form>`;
    
    document.getElementById("contenedor-test").innerHTML = html;
    
    // Event listeners
    document.getElementById("btn-desmarcar").addEventListener("click", () => {
      document.querySelectorAll('input[name="respuesta"]:checked').forEach(el => el.checked = false);
      respuestasUsuario[i] = null;
    });
    
    if (i > 0) {
      document.getElementById("btn-anterior").addEventListener("click", () => {
        guardarRespuesta(i);
        mostrarPregunta(i - 1);
      });
    }
    
    document.getElementById("form-pregunta").addEventListener("submit", function(e) {
      e.preventDefault();
      guardarRespuesta(i);
      
      if (i + 1 < preguntas.length) {
        mostrarPregunta(i + 1);
      } else {
        mostrarConfirmacionFinalizacion();
      }
    });
    
    document.getElementById("btn-finalizar").style.display = "block";
  }
  
  function guardarRespuesta(indice) {
    const seleccion = document.querySelector('input[name="respuesta"]:checked');
    respuestasUsuario[indice] = seleccion ? seleccion.value : null;
  }

  function mostrarConfirmacionFinalizacion() {
    const sinContestar = respuestasUsuario.filter(r => r === null).length;
    Swal.fire({
      title: '¿Finalizar test?',
      html: sinContestar > 0 
        ? `Tienes <b>${sinContestar}</b> pregunta${sinContestar > 1 ? 's' : ''} sin contestar.<br>¿Seguro que quieres finalizar?`
        : '¿Quieres finalizar el test y ver los resultados?',
      icon: 'question',
      showCancelButton: true,
      confirmButtonText: 'Sí, finalizar',
      cancelButtonText: 'Seguir revisando',
      confirmButtonColor: '#2c6fbb',
      cancelButtonColor: '#6c757d'
    }).then((result) => {
      if (result.isConfirmed) {
        mostrarResultados();
      }
    });
  }

  async function mostrarResultados() {
    clearInterval(intervaloTemporizador);
    document.getElementById("contenedor-test").style.display = "none";
    document.getElementById("btn-finalizar").style.display = "none";
    document.getElementById("barra-progreso-contenedor").style.display = "none";
    
    const cont = document.getElementById("contenedor-resultados");
    cont.style.display = "block";
    
    let aciertos = 0;
    let resultadosHTML = "";
    let statsPorTema = {};
    
    // Procesar resultados
    preguntas.forEach((p, i) => {
      const correcta = p.respuesta_correcta;
      const seleccion = respuestasUsuario[i];
      const explicacion = p.explicacion || "No hay explicación disponible.";
      const tema = p.tema || "General";
      
      // Registrar tema
      temasTest.add(tema);
      if (!statsPorTema[tema]) {
        statsPorTema[tema] = { total: 0, aciertos: 0 };
      }
      statsPorTema[tema].total++;
      
      // Determinar clase CSS según resultado
      let estadoClase = "";
      if (seleccion === correcta) {
        estadoClase = "acierto";
        aciertos++;
        statsPorTema[tema].aciertos++;
      } else if (seleccion === null) {
        estadoClase = "blanco";
      } else {
        estadoClase = "fallo";
      }
      
      // Generar HTML de opciones con estilos
      let opcionesHTML = "";
      for (const letra in p.opciones) {
        const opcion = p.opciones[letra];
        let opcionClase = "";
        let icono = "";
        
        if (letra === correcta) {
          opcionClase = "opcion-correcta";
          icono = "✓";
        } else if (letra === seleccion && letra !== correcta) {
          opcionClase = "opcion-incorrecta";
          icono = "✗";
        }
        
        opcionesHTML += `
          <div class="opcion ${opcionClase}">
            <input type="radio" disabled ${seleccion === letra ? 'checked' : ''}>
            <label>${letra}) ${opcion} ${icono ? `<span style="margin-left:10px;">${icono}</span>` : ''}</label>
          </div>`;
      }
      
      // HTML de la pregunta
      resultadosHTML += `
        <div class="pregunta-resultado ${estadoClase}">
          <div class="pregunta-texto">${p.pregunta}</div>
          <div class="opciones-container">${opcionesHTML}</div>
          <div class="explicacion">
            <strong>Explicación:</strong> ${explicacion}
          </div>
        </div>`;
    });
    
    // Calcular estadísticas
    const totalPreguntas = preguntas.length;
    const fallos = totalPreguntas - aciertos - respuestasUsuario.filter(r => r === null).length;
    const sinResponder = respuestasUsuario.filter(r => r === null).length;
    const porcentaje = ((aciertos / totalPreguntas) * 100).toFixed(1);
    const nota = (aciertos * 1 - fallos * 0.33).toFixed(2);
    const notaEquivalente = ((nota / totalPreguntas) * 70).toFixed(2);
    const segundosTotales = Math.floor((Date.now() - tiempoInicio) / 1000);
    
    // Generar HTML de estadísticas
    let estadisticasHTML = `
      <div class="resumen-test">
        <h2 style="text-align:center; margin-bottom:25px;">📊 Resultados del Test</h2>
        
        <div class="estadisticas-container">
          <div class="estadistica acierto">
            <div class="estadistica-numero">${aciertos}</div>
            <div>✅ Aciertos</div>
            <div>${(aciertos/totalPreguntas*100).toFixed(1)}%</div>
          </div>
          
          <div class="estadistica fallo">
            <div class="estadistica-numero">${fallos}</div>
            <div>❌ Fallos</div>
            <div>${(fallos/totalPreguntas*100).toFixed(1)}%</div>
          </div>
          
          <div class="estadistica blanco">
            <div class="estadistica-numero">${sinResponder}</div>
            <div>⏸ Sin responder</div>
            <div>${(sinResponder/totalPreguntas*100).toFixed(1)}%</div>
          </div>
        </div>
        
        <div class="nota-simulada">
          <div>Tu puntuación</div>
          <div class="nota-equivalente">${notaEquivalente}/70</div>
          <div>Equivalente a ${nota}/${totalPreguntas} (${porcentaje}%)</div>
        </div>
        
        <div>
          <div style="display:flex; justify-content:space-between; margin-bottom:8px;">
            <span>Porcentaje de aciertos:</span>
            <span><strong>${porcentaje}%</strong></span>
          </div>
          <div class="barra-porcentaje">
            <div class="relleno-porcentaje" style="width:${porcentaje}%"></div>
          </div>
        </div>
      </div>`;
    
    // Generar HTML de temas si existen
    let temasHTML = '';
    if (temasTest.size > 0) {
      temasHTML += `<div class="detalles-tema">
        <h3>📚 Rendimiento por temas</h3>
        <div class="temas-grid">`;
      
      temasTest.forEach(tema => {
        const stat = statsPorTema[tema];
        const porcentajeTema = stat.total > 0 ? ((stat.aciertos / stat.total) * 100).toFixed(1) : 0;
        
        temasHTML += `
          <div class="tema-card">
            <div><strong>${tema}</strong></div>
            <div>${stat.aciertos}/${stat.total} (${porcentajeTema}%)</div>
            <div class="tema-porcentaje">
              <div class="tema-relleno" style="width:${porcentajeTema}%"></div>
            </div>
          </div>`;
      });
      
      temasHTML += `</div></div>`;
    }
    
    // HTML final
    cont.innerHTML = `
      ${estadisticasHTML}
      ${temasHTML}
      <h3 style="margin:30px 0 15px;">Detalle de preguntas:</h3>
      ${resultadosHTML}
      <div style="text-align:center; margin-top:30px;">
        <button id="btn-repetir" class="btn btn-siguiente">🔁 Repetir Test</button>
      </div>`;
    
    // Event listener para repetir test
    document.getElementById("btn-repetir").addEventListener("click", () => {
      location.reload();
    });
    
    // Guardar resultados en el servidor
    try {
      await fetch("https://oposicion-age.onrender.com/guardar-test", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          usuario_id: usuarioEmail,
          contenido: preguntas,
          respuestas: respuestasUsuario,
          metadatos: {
            tiempo: segundosTotales,
            tipo: "repetido",
            temas: Array.from(temasTest),
            aciertos: aciertos,
            total: totalPreguntas
          }
        })
      });
      console.log("✅ Test guardado como repetido");
    } catch (err) {
      console.error("❌ Error al guardar test:", err);
    }
  }

  document.getElementById("btn-finalizar").addEventListener("click", mostrarConfirmacionFinalizacion);

  window.addEventListener("load", async () => {
    try {
      const res = await fetch(`https://oposicion-age.onrender.com/ultimo-test?usuario_id=${usuarioEmail}`);
      const datos = await res.json();

      if (!datos.test || datos.test.length === 0) {
        document.getElementById("contenedor-test").innerHTML = `
          <div style="text-align:center; padding:40px;">
            <h3>No se encontró test anterior</h3>
            <p>No hay ningún test disponible para repetir.</p>
          </div>`;
        return;
      }

      preguntas = datos.test;
      respuestasUsuario = Array(preguntas.length).fill(null);
      indicePreguntaActual = 0;
      
      // Extraer temas únicos
      preguntas.forEach(p => {
        if (p.tema) temasTest.add(p.tema);
      });
      
      iniciarTemporizador();
      mostrarPregunta(indicePreguntaActual);
    } catch (err) {
      console.error("Error:", err);
      document.getElementById("contenedor-test").innerHTML = `
        <div style="text-align:center; padding:40px; color:#ef5350;">
          <h3>Error al cargar el test</h3>
          <p>Por favor intenta nuevamente más tarde.</p>
        </div>`;
    }
  });
  </script>
