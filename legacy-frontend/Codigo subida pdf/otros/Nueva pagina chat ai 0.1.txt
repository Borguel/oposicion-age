    <script>
        let currentAction = '';
        let selectedFile = null;

        function openModal(action) {
            currentAction = action;
            const modal = document.getElementById('pdfModal');
            const title = document.getElementById('modalTitle');
            
            // Configurar título según la acción
            const titles = {
                'resumir': 'Resumir Documento PDF',
                'esquema': 'Generar Esquema desde PDF', 
                'test': 'Crear Test desde PDF',
                'tarjetas': 'Generar Tarjetas de Memoria'
            };
            
            title.textContent = titles[action] || 'Subir Documento PDF';
            modal.style.display = 'flex';
            
            // Resetear estado
            resetModal();
        }

        function closeModal() {
            document.getElementById('pdfModal').style.display = 'none';
            resetModal();
        }

        function resetModal() {
            selectedFile = null;
            document.getElementById('pdfFile').value = '';
            document.getElementById('fileName').textContent = '';
            document.getElementById('processBtn').disabled = true;
            document.getElementById('loading').style.display = 'none';
            document.getElementById('result').style.display = 'none';
        }

        function handleFileSelect(input) {
            if (input.files.length > 0) {
                selectedFile = input.files[0];
                document.getElementById('fileName').textContent = selectedFile.name;
                document.getElementById('processBtn').disabled = false;
            }
        }

        async function processDocument() {
            if (!selectedFile) return;

            const loading = document.getElementById('loading');
            const processBtn = document.getElementById('processBtn');
            const resultDiv = document.getElementById('result');
            const resultContent = document.getElementById('resultContent');

            // Mostrar loading
            loading.style.display = 'block';
            processBtn.disabled = true;
            resultDiv.style.display = 'none';

            try {
                const formData = new FormData();
                formData.append('pdf', selectedFile);

                // Determinar endpoint según la acción
                const endpoints = {
                    'resumir': '/resumir-pdf',
                    'esquema': '/generar-esquema-desde-pdf', // Necesitarás crear esta ruta
                    'test': '/generar-test-desde-pdf', // Necesitarás crear esta ruta  
                    'tarjetas': '/generar-tarjetas-desde-pdf' // Necesitarás crear esta ruta
                };

                const endpoint = endpoints[currentAction];
                
                const response = await fetch(endpoint, {
                    method: 'POST',
                    body: formData
                });

                if (!response.ok) {
                    throw new Error('Error en el servidor');
                }

                const data = await response.json();
                
                // Mostrar resultado
                resultContent.innerHTML = formatResult(data, currentAction);
                resultDiv.style.display = 'block';

            } catch (error) {
                console.error('Error:', error);
                resultContent.innerHTML = `<p style="color: red;">Error al procesar el documento: ${error.message}</p>`;
                resultDiv.style.display = 'block';
            } finally {
                loading.style.display = 'none';
            }
        }

        function formatResult(data, action) {
            switch(action) {
                case 'resumir':
                    return `<div class="resumen">${data.resumen.replace(/\n/g, '<br>')}</div>`;
                
                case 'esquema':
                    return `<div class="esquema">${data.esquema.replace(/\n/g, '<br>')}</div>`;
                
                case 'test':
                    if (data.test && data.test.length > 0) {
                        return data.test.map((pregunta, index) => `
                            <div class="pregunta" style="margin-bottom: 15px; padding: 10px; border-left: 3px solid #4f46e5;">
                                <strong>Pregunta ${index + 1}:</strong> ${pregunta.pregunta}<br>
                                <strong>Opciones:</strong><br>
                                ${Object.entries(pregunta.opciones).map(([key, value]) => 
                                    `${key}) ${value}`).join('<br>')}<br>
                                <strong>Respuesta correcta:</strong> ${pregunta.respuesta_correcta}<br>
                                <em>Explicación:</em> ${pregunta.explicacion}
                            </div>
                        `).join('');
                    }
                    return '<p>No se pudieron generar preguntas</p>';
                
                case 'tarjetas':
                    if (data.tarjetas && data.tarjetas.length > 0) {
                        return data.tarjetas.map((tarjeta, index) => `
                            <div class="tarjeta" style="margin-bottom: 10px; padding: 10px; background: #f8fafc; border-radius: 5px;">
                                <strong>${index + 1}. ${tarjeta.pregunta}</strong><br>
                                <span style="color: #059669;">→ ${tarjeta.respuesta}</span>
                            </div>
                        `).join('');
                    }
                    return '<p>No se pudieron generar tarjetas</p>';
                
                default:
                    return '<p>Resultado no formateado</p>';
            }
        }

        // Cerrar modal al hacer clic fuera
        window.onclick = function(event) {
            const modal = document.getElementById('pdfModal');
            if (event.target === modal) {
                closeModal();
            }
        }
