    <script>
        // ✅ Usa tu URL de Render
        const BACKEND_URL = "https://oposicion-age.onrender.com";
        
        // Elementos del DOM
        const uploadArea = document.getElementById('upload-area');
        const pdfFileInput = document.getElementById('pdf-file');
        const resumenContainer = document.getElementById('resumen-container');
        const resumenContent = document.getElementById('resumen-content');
        const chatBox = document.getElementById('chat-box');
        const userInput = document.getElementById('user-input');
        const sendBtn = document.getElementById('send-btn');
        
        // Evento para abrir el selector de archivos al hacer clic en el área de carga
        uploadArea.addEventListener('click', () => {
            pdfFileInput.click();
        });
        
        // Evento para arrastrar y soltar archivos
        uploadArea.addEventListener('dragover', (e) => {
            e.preventDefault();
            uploadArea.classList.add('drag-over');
        });
        
        uploadArea.addEventListener('dragleave', () => {
            uploadArea.classList.remove('drag-over');
        });
        
        uploadArea.addEventListener('drop', (e) => {
            e.preventDefault();
            uploadArea.classList.remove('drag-over');
            
            if (e.dataTransfer.files.length) {
                pdfFileInput.files = e.dataTransfer.files;
                subirPDF();
            }
        });
        
        // Evento para cuando se selecciona un archivo
        pdfFileInput.addEventListener('change', subirPDF);
        
        // Función para subir y resumir PDF
        async function subirPDF() {
            const file = pdfFileInput.files[0];
            
            if (!file) {
                alert('Por favor selecciona un archivo PDF');
                return;
            }
            
            const formData = new FormData();
            formData.append('pdf', file);
            
            try {
                // Mostrar mensaje de espera
                resumenContent.innerHTML = '<div class="spinner"></div> <span>Procesando documento...</span>';
                resumenContainer.style.display = 'flex';
                
                // Simulación de carga
                setTimeout(async () => {
                    try {
                        const response = await fetch(`${BACKEND_URL}/resumir-pdf`, {
                            method: 'POST',
                            body: formData
                        });
                        
                        const data = await response.json();
                        
                        if (data.resumen) {
                            // Formatear el resumen con saltos de línea
                            const formattedResumen = data.resumen.replace(/\n/g, '<br>');
                            resumenContent.innerHTML = formattedResumen;
                            
                            // Añadir mensaje al chat
                            addMessageToChat("He subido un documento PDF para que lo resumas.", 'user');
                            
                            // Simular respuesta del asistente
                            setTimeout(() => {
                                addMessageToChat("¡Documento procesado con éxito! He creado un resumen del PDF que subiste. ¿Te gustaría que te explique algún punto en particular o que genere un test basado en este contenido?", 'bot');
                            }, 1500);
                        } else {
                            resumenContent.innerHTML = "Error: " + (data.error || 'Error desconocido');
                        }
                    } catch (error) {
                        resumenContent.innerHTML = "Error de conexión: " + error.message;
                    }
                }, 2000);
            } catch (error) {
                resumenContent.innerHTML = "Error: " + error.message;
            }
        }
        
        // Función para enviar mensajes
        function enviarMensaje() {
            const message = userInput.value.trim();
            
            if (!message) return;
            
            // Añadir mensaje de usuario al chat
            addMessageToChat(message, 'user');
            
            // Limpiar el input
            userInput.value = '';
            
            // Mostrar estado de "escribiendo"
            showTypingIndicator();
            
            // Simular respuesta después de un retraso
            setTimeout(() => {
                // Ocultar indicador de escritura
                hideTypingIndicator();
                
                // Generar respuesta basada en el mensaje del usuario
                let response = "Gracias por tu pregunta. Estoy procesando tu consulta y te responderé en breve.";
                
                if (message.toLowerCase().includes('test')) {
                    response = "¡Claro! Puedo generar un test personalizado para ti. ¿Sobre qué temas específicos te gustaría que fuera el test? Puedo crearlo basado en los documentos que has subido o en el temario general.";
                } else if (message.toLowerCase().includes('esquema')) {
                    response = "Excelente idea. Los esquemas son muy útiles para estudiar. ¿De qué tema te gustaría que prepare un esquema? Puedo hacerlo detallado o resumido, según tus preferencias.";
                } else if (message.toLowerCase().includes('temario')) {
                    response = "El temario para Administrativo del Estado consta de 3 bloques principales:<br><br>1. Organización del Estado y Administración Electrónica<br>2. Gestión de Personal y Derecho Administrativo<br>3. Gestión Financiera<br><br>¿Quieres que te detalle algún bloque específico?";
                } else if (message.toLowerCase().includes('ayuda') || message.toLowerCase().includes('qué puedes hacer')) {
                    response = "Puedo ayudarte con:<br><br>- Resumir documentos PDF<br>- Explicar temas del temario<br>- Generar tests personalizados<br>- Crear esquemas de estudio<br>- Resolver dudas sobre procedimientos administrativos<br><br>Solo dime qué necesitas 😊";
                } else {
                    response = "Entiendo tu consulta sobre \"" + message + "\". Como asistente especializado en oposiciones para Administrativo del Estado, puedo darte información precisa sobre este tema. ¿Te gustaría que profundice en algún aspecto en particular o que te proporcione ejemplos prácticos?";
                }
                
                addMessageToChat(response, 'bot');
            }, 2000);
        }
        
        // Función para añadir mensajes al chat
        function addMessageToChat(message, sender) {
            const messageElement = document.createElement('div');
            messageElement.classList.add('message');
            messageElement.classList.add(sender === 'user' ? 'user-message' : 'bot-message');
            
            const now = new Date();
            const timeString = now.getHours() + ':' + (now.getMinutes() < 10 ? '0' : '') + now.getMinutes();
            
            messageElement.innerHTML = `
                <div class="message-header">
                    <i class="fas ${sender === 'user' ? 'fa-user' : 'fa-robot'}"></i>
                    <span>${sender === 'user' ? 'Tú' : 'Asistente IA'}</span>
                </div>
                <div class="message-content">
                    <p>${message}</p>
                </div>
                <div class="message-time">${timeString}</div>
            `;
            
            chatBox.appendChild(messageElement);
            chatBox.scrollTop = chatBox.scrollHeight;
        }
        
        // Función para mostrar indicador de "escribiendo"
        function showTypingIndicator() {
            const typingElement = document.createElement('div');
            typingElement.classList.add('message', 'bot-message');
            typingElement.id = 'typing-indicator';
            typingElement.innerHTML = `
                <div class="message-header">
                    <i class="fas fa-robot"></i>
                    <span>Asistente IA</span>
                </div>
                <div class="message-content">
                    <div class="spinner"></div> <span>Escribiendo...</span>
                </div>
            `;
            
            chatBox.appendChild(typingElement);
            chatBox.scrollTop = chatBox.scrollHeight;
        }
        
        // Función para ocultar indicador de "escribiendo"
        function hideTypingIndicator() {
            const typingElement = document.getElementById('typing-indicator');
            if (typingElement) {
                typingElement.remove();
            }
        }
        
        // Event listeners
        sendBtn.addEventListener('click', enviarMensaje);
        userInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                enviarMensaje();
            }
        });
