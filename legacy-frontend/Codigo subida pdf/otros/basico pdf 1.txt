        .container { max-width: 800px; margin: 0 auto; padding: 20px; }
        .chat-box { height: 400px; border: 1px solid #ccc; overflow-y: auto; padding: 10px; margin-bottom: 10px; }
        .message { margin-bottom: 10px; padding: 8px; border-radius: 8px; }
        .user-message { background-color: #e3f2fd; text-align: right; }
        .bot-message { background-color: #f5f5f5; }
        .spinner {
            border: 4px solid rgba(0, 0, 0, 0.1);
            border-radius: 50%;
            border-top: 4px solid #3498db;
            width: 20px;
            height: 20px;
            animation: spin 1s linear infinite;
            display: inline-block;
            vertical-align: middle;
            margin-right: 10px;
        }
        @keyframes spin {
            0% { transform: rotate(0deg); }
            100% { transform: rotate(360deg); }
        }

<body>
    <div class="container">
        <h1>Asistente para Opositores</h1>
        
        <h2>Subir PDF</h2>
        <input type="file" id="pdf-file" accept=".pdf">
        <button onclick="subirPDF()">Resumir PDF</button>
        <div id="resumen-container" style="margin-top: 20px; display: none;">
            <h3>Resumen:</h3>
            <div id="resumen-content"></div>
        </div>
        
        <h2>Chat con DeepSeek</h2>
        <div class="chat-box" id="chat-box"></div>
        <input type="text" id="user-input" placeholder="Escribe tu pregunta...">
        <button onclick="enviarMensaje()">Enviar</button>
    </div>

    <script>
        // ✅ Usa tu URL de Render
        const BACKEND_URL = "https://oposicion-age.onrender.com";
        
        // Subir y resumir PDF
        async function subirPDF() {
            const fileInput = document.getElementById('pdf-file');
            const file = fileInput.files[0];
            
            if (!file) {
                alert('Por favor selecciona un archivo PDF');
                return;
            }
            
            const formData = new FormData();
            formData.append('pdf', file);
            
            try {
                // Mostrar mensaje de espera
                document.getElementById('resumen-content').innerHTML = '<div class="spinner"></div> Procesando PDF...';
                document.getElementById('resumen-container').style.display = 'block';
                
                const response = await fetch(`${BACKEND_URL}/resumir-pdf`, {
                    method: 'POST',
                    body: formData
                });
                
                const data = await response.json();
                
                if (data.resumen) {
                    document.getElementById('resumen-content').innerHTML = 
                        data.resumen.replace(/\n/g, '<br>');
                } else {
                    document.getElementById('resumen-content').innerHTML = 
                        "Error: " + (data.error || 'Error desconocido');
                }
            } catch (error) {
                document.getElementById('resumen-content').innerHTML = 
                    "Error de conexión: " + error.message;
            }
        }
        
        // Chat con DeepSeek
        async function enviarMensaje() {
            const input = document.getElementById('user-input');
            const message = input.value.trim();
            
            if (!message) return;
            
            // Mostrar mensaje de usuario
            const chatBox = document.getElementById('chat-box');
            chatBox.innerHTML += `<div class="message user-message">${message}</div>`;
            input.value = '';
            
            try {
                // Mostrar indicador de que el bot está pensando
                chatBox.innerHTML += `<div class="message bot-message"><div class="spinner"></div> Pensando...</div>`;
                chatBox.scrollTop = chatBox.scrollHeight;
                
                const response = await fetch(`${BACKEND_URL}/chat-deepseek`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ mensaje: message })
                });
                
                const data = await response.json();
                
                // Eliminar el mensaje "Pensando..."
                chatBox.removeChild(chatBox.lastChild);
                
                if (data.respuesta) {
                    chatBox.innerHTML += `<div class="message bot-message">${data.respuesta}</div>`;
                } else {
                    chatBox.innerHTML += `<div class="message bot-message">Error: ${data.error || 'Respuesta vacía'}</div>`;
                }
                
                // Scroll al final
                chatBox.scrollTop = chatBox.scrollHeight;
            } catch (error) {
                // Eliminar el mensaje "Pensando..." y mostrar error
                chatBox.removeChild(chatBox.lastChild);
                chatBox.innerHTML += `<div class="message bot-message">Error de conexión: ${error.message}</div>`;
                chatBox.scrollTop = chatBox.scrollHeight;
            }
        }
    </script>
</body>
