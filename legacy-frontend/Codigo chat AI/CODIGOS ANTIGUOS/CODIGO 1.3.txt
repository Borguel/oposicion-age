<script>
// Lógica para la X fija (sale del chat)
document.addEventListener("DOMContentLoaded", function() {
  var cerrarChat = document.getElementById("cerrar-chat-total");
  if(cerrarChat){
    cerrarChat.addEventListener("click", function() {
      window.location.href = "https://lightslategray-caribou-622401.hostingersite.com/panel-de-usuario/";
    });
  }

  // ---- RESTO DEL JS DEL CHAT ----
  const input = document.getElementById("chat-input");
  const sendBtn = document.getElementById("chat-send");
  const chatMessages = document.getElementById("chat-messages");
  const toggleSidebar = document.getElementById("toggle-sidebar");
  const chatSidebar = document.getElementById("chat-sidebar");
  const chatHistory = document.getElementById("chat-history");

  // Abrir/cerrar sidebar
  toggleSidebar.addEventListener("click", () => {
    chatSidebar.classList.toggle("visible");
  });

  // Cierra sidebar al pulsar fuera
  document.addEventListener("click", (e) => {
    if (
      window.innerWidth <= 900 &&
      chatSidebar.classList.contains("visible") &&
      !chatSidebar.contains(e.target) &&
      e.target !== toggleSidebar
    ) {
      chatSidebar.classList.remove("visible");
    }
  });

  // Mensajes visuales pro
  function agregarMensaje(tipo, texto) {
    const div = document.createElement("div");
    if (tipo === "user") {
      div.className = "mensaje-user";
      div.innerHTML = `<div class="bubble-user">${texto}</div>`;
    } else {
      div.className = "mensaje-bot";
      div.innerHTML = `
        <div class="avatar-bot-mini">
          <img src="https://randomuser.me/api/portraits/women/44.jpg" alt="Bot" />
        </div>
        <div class="bubble-bot anim-fade-in">${texto}</div>
      `;
    }
    chatMessages.appendChild(div);
    chatMessages.scrollTop = chatMessages.scrollHeight;
  }

  // Envía mensaje
  function enviarMensaje() {
    const texto = input.value.trim();
    if (!texto) return;
    agregarMensaje("user", texto);
    input.value = "";

    fetch("https://oposicion-age.onrender.com/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        mensaje: texto,
        temas: [],
        usuario_id: typeof usuarioEmail !== "undefined" ? usuarioEmail : "anonimo",
        chat_id: chatIdActual
      })
    })
      .then(res => res.json())
      .then(data => {
        agregarMensaje("bot", data.respuesta || "Sin respuesta.");
        chatIdActual = data.chat_id;
        cargarHistorial();
      })
      .catch(() => {
        agregarMensaje("bot", "❌ Error al conectar.");
      });
  }

  // Historial
  let conversaciones = [];
  let chatIdActual = null;

  function cargarHistorial() {
    const id = typeof usuarioEmail !== "undefined" ? usuarioEmail : "anonimo";
    fetch(`https://oposicion-age.onrender.com/conversaciones?usuario_id=${id}`)
      .then(res => res.json())
      .then(data => {
        conversaciones = data.conversaciones || [];
        mostrarHistorialEnSidebar();
      })
      .catch(err => {
        console.error("❌ Error al cargar historial:", err);
      });
  }

  function mostrarHistorialEnSidebar() {
    chatHistory.innerHTML = "";
    const nueva = document.createElement("li");
    nueva.textContent = "➕ Nueva conversación";
    nueva.onclick = iniciarNuevaConversacion;
    chatHistory.appendChild(nueva);

    conversaciones.forEach((conv) => {
      const li = document.createElement("li");
      li.textContent = conv.titulo || "Conversación";
      li.dataset.id = conv.id;
      li.onclick = () => cargarConversacion(conv.id);
      chatHistory.appendChild(li);
    });
  }

  function cargarConversacion(id) {
    chatMessages.innerHTML = "";
    chatIdActual = id;

    fetch(`https://oposicion-age.onrender.com/conversacion/${id}?usuario_id=${typeof usuarioEmail !== "undefined" ? usuarioEmail : "anonimo"}`)
      .then(res => res.json())
      .then(data => {
        if (data.mensajes) {
          data.mensajes.forEach(m => {
            agregarMensaje(m.role === "user" ? "user" : "bot", m.content);
          });
        }
      });

    document.querySelectorAll("#chat-history li").forEach(li => {
      li.classList.remove("active");
      if (li.dataset.id === id) li.classList.add("active");
    });
  }

  function iniciarNuevaConversacion() {
    chatMessages.innerHTML = "";
    chatIdActual = null;
  }

  // Eventos input
  sendBtn.addEventListener("click", enviarMensaje);
  input.addEventListener("keydown", function (e) {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      enviarMensaje();
    }
  });

  window.addEventListener("DOMContentLoaded", cargarHistorial);

  // Salir del chat (sidebar)
  document.getElementById("salir-del-chat-sidebar").addEventListener("click", function() {
    window.location.href = "https://lightslategray-caribou-622401.hostingersite.com/panel-de-usuario/";
  });
});
</script>
