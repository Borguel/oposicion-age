    </div>

    <script>
        document.addEventListener("DOMContentLoaded", function() {
            // Elementos del DOM
            const input = document.getElementById("chat-input");
            const sendBtn = document.getElementById("chat-send");
            const chatMessages = document.getElementById("chat-messages");
            const toggleSidebar = document.getElementById("toggle-sidebar");
            const chatSidebar = document.getElementById("chat-sidebar");
            const chatHistory = document.getElementById("chat-history");
            const cerrarChatTotal = document.getElementById("cerrar-chat-total");
            const newChatBtn = document.getElementById("new-chat");
            const toggleHistoryBtn = document.getElementById("toggle-history");
            const newChatSidebar = document.getElementById("nueva-conversacion-sidebar");
            
            // URLs
            const exitUrl = "https://lightslategray-caribou-622401.hostingersite.com/panel-de-usuario/";
            
            // Funciones de salida
            function salirDelChat() {
                window.location.href = exitUrl;
            }
            
            // Event listeners para botones de salida
            cerrarChatTotal.addEventListener("click", salirDelChat);
            
            // Abrir/cerrar sidebar
            toggleSidebar.addEventListener("click", () => {
                chatSidebar.classList.toggle("visible");
            });
            
            // Alternar visibilidad del historial
            toggleHistoryBtn.addEventListener("click", () => {
                const icon = toggleHistoryBtn.querySelector('i');
                if (chatSidebar.style.width === "0px" || chatSidebar.style.width === "") {
                    chatSidebar.style.width = "280px";
                    icon.className = "fas fa-chevron-left";
                } else {
                    chatSidebar.style.width = "0";
                    icon.className = "fas fa-chevron-right";
                }
            });
            
            // Variables para gestionar las conversaciones
            let conversaciones = [];
            let chatIdActual = null;
            // Asumimos que tienes una variable global `usuarioEmail` definida en otro script
            // Si no, puedes obtenerla de alguna otra manera (por ejemplo, desde un input oculto o desde el backend)

            // Función para agregar mensajes al chat
            function agregarMensaje(tipo, texto) {
                const ahora = new Date();
                const hora = ahora.getHours().toString().padStart(2, '0');
                const minutos = ahora.getMinutes().toString().padStart(2, '0');
                const tiempo = `${hora}:${minutos}`;
                
                const div = document.createElement("div");
                
                if (tipo === "user") {
                    div.className = "mensaje-user";
                    div.innerHTML = `
                        <div class="bubble-container">
                            <div class="bubble-user">${texto}</div>
                            <div class="message-time">${tiempo}</div>
                        </div>
                    `;
                } else {
                    div.className = "mensaje-bot";
                    div.innerHTML = `
                        <div class="avatar-bot-mini">
                            <img src="https://randomuser.me/api/portraits/women/44.jpg" alt="Bot" />
                        </div>
                        <div class="bubble-container">
                            <div class="bubble-bot">${texto}</div>
                            <div class="message-time">${tiempo}</div>
                        </div>
                    `;
                }
                
                chatMessages.appendChild(div);
                chatMessages.scrollTop = chatMessages.scrollHeight;
            }

            // Función para mostrar el indicador de que el bot está escribiendo
            function mostrarTypingIndicator() {
                const typingIndicator = document.createElement('div');
                typingIndicator.className = 'mensaje-bot';
                typingIndicator.innerHTML = `
                    <div class="avatar-bot-mini">
                        <img src="https://randomuser.me/api/portraits/women/44.jpg" alt="Bot" />
                    </div>
                    <div class="bot-typing">
                        <div class="typing-dot"></div>
                        <div class="typing-dot"></div>
                        <div class="typing-dot"></div>
                    </div>
                `;
                chatMessages.appendChild(typingIndicator);
                chatMessages.scrollTop = chatMessages.scrollHeight;
                return typingIndicator;
            }

            // Enviar mensaje al backend
            function enviarMensaje() {
                const texto = input.value.trim();
                if (!texto) return;
                
                // Agregar mensaje de usuario
                agregarMensaje("user", texto);
                input.value = "";
                input.style.height = 'auto';
                
                // Mostrar indicador de typing
                const typingIndicator = mostrarTypingIndicator();
                
                // Datos para enviar
                const data = {
                    mensaje: texto,
                    temas: [],
                    usuario_id: typeof usuarioEmail !== "undefined" ? usuarioEmail : "anonimo",
                    chat_id: chatIdActual
                };
                
                // Llamada al backend
                fetch("https://oposicion-age.onrender.com/chat", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(data)
                })
                .then(res => res.json())
                .then(data => {
                    // Eliminar el indicador de typing
                    chatMessages.removeChild(typingIndicator);
                    
                    // Agregar la respuesta del bot
                    agregarMensaje("bot", data.respuesta || "No pude obtener una respuesta.");
                    
                    // Actualizar el chatIdActual si es una nueva conversación
                    if (data.chat_id) {
                        chatIdActual = data.chat_id;
                    }
                    
                    // Recargar el historial
                    cargarHistorial();
                })
                .catch(() => {
                    chatMessages.removeChild(typingIndicator);
                    agregarMensaje("bot", "❌ Error al conectar con el servidor.");
                });
            }

            // Cargar el historial de conversaciones
            function cargarHistorial() {
                const usuario_id = typeof usuarioEmail !== "undefined" ? usuarioEmail : "anonimo";
                fetch(`https://oposicion-age.onrender.com/conversaciones?usuario_id=${usuario_id}`)
                .then(res => res.json())
                .then(data => {
                    conversaciones = data.conversaciones || [];
                    mostrarHistorialEnSidebar();
                })
                .catch(err => {
                    console.error("Error al cargar historial:", err);
                });
            }

            // Mostrar el historial en el sidebar
            function mostrarHistorialEnSidebar() {
                chatHistory.innerHTML = "";
                
                // Botón para nueva conversación
                const nueva = document.createElement("li");
                nueva.textContent = "➕ Nueva conversación";
                nueva.onclick = iniciarNuevaConversacion;
                chatHistory.appendChild(nueva);
                
                // Listar todas las conversaciones
                conversaciones.forEach(conv => {
                    const li = document.createElement("li");
                    li.textContent = conv.titulo || "Conversación";
                    li.dataset.id = conv.id;
                    li.onclick = () => cargarConversacion(conv.id);
                    chatHistory.appendChild(li);
                    
                    // Si es la conversación actual, marcarla como activa
                    if (chatIdActual === conv.id) {
                        li.classList.add("active");
                    }
                });
            }

            // Cargar una conversación específica
            function cargarConversacion(id) {
                chatMessages.innerHTML = "";
                chatIdActual = id;
                
                const usuario_id = typeof usuarioEmail !== "undefined" ? usuarioEmail : "anonimo";
                fetch(`https://oposicion-age.onrender.com/conversacion/${id}?usuario_id=${usuario_id}`)
                .then(res => res.json())
                .then(data => {
                    if (data.mensajes) {
                        data.mensajes.forEach(m => {
                            agregarMensaje(m.role === "user" ? "user" : "bot", m.content);
                        });
                    }
                });
                
                // Actualizar el elemento activo en el historial
                document.querySelectorAll("#chat-history li").forEach(li => {
                    li.classList.remove("active");
                    if (li.dataset.id === id) {
                        li.classList.add("active");
                    }
                });
            }

            // Iniciar una nueva conversación
            function iniciarNuevaConversacion() {
                chatMessages.innerHTML = "";
                chatIdActual = null;
                
                // Mostrar mensaje de bienvenida
                showWelcomeMessage();
                
                // Desmarcar cualquier conversación activa
                document.querySelectorAll("#chat-history li").forEach(li => {
                    li.classList.remove("active");
                });
            }

            // Mostrar mensaje de bienvenida
            function showWelcomeMessage() {
                chatMessages.innerHTML = `
                    <div class="welcome-message">
                        <div class="welcome-title">¡Hola, opositor/a! 👋</div>
                        <div class="welcome-subtitle">Soy tu asistente especializado en la oposición de Administrativo del Estado. ¿En qué puedo ayudarte hoy?</div>
                        
                        <div class="suggestions-container">
                            <div class="suggestions-title">
                                <i class="fas fa-lightbulb"></i> Preguntas frecuentes para opositores:
                            </div>
                            <div class="suggestions-grid">
                                <div class="suggestion-item">¿Cuál es la estructura del examen de administrativo?</div>
                                <div class="suggestion-item">Explícame el procedimiento administrativo común</div>
                                <div class="suggestion-item">¿Qué temas son más importantes para el primer ejercicio?</div>
                                <div class="suggestion-item">¿Cómo funciona el sistema de selección por concurso-oposición?</div>
                            </div>
                        </div>
                    </div>
                `;
                
                // Agregar eventos a las sugerencias
                document.querySelectorAll('.suggestion-item').forEach(item => {
                    item.addEventListener('click', function() {
                        input.value = this.textContent;
                        input.focus();
                        input.dispatchEvent(new Event('input', { bubbles: true }));
                    });
                });
            }
            
            // Eventos para enviar mensaje
            sendBtn.addEventListener("click", enviarMensaje);
            input.addEventListener("keydown", function (e) {
                if (e.key === "Enter" && !e.shiftKey) {
                    e.preventDefault();
                    enviarMensaje();
                }
            });
            
            // Autoajustar altura del textarea
            input.addEventListener('input', function() {
                this.style.height = 'auto';
                this.style.height = (this.scrollHeight) + 'px';
            });
            
            // Nueva conversación desde diferentes botones
            newChatBtn.addEventListener("click", iniciarNuevaConversacion);
            newChatSidebar.addEventListener("click", iniciarNuevaConversacion);
            
            // Cargar el historial al inicio
            cargarHistorial();
        });
    </script>
</body>
